# ppq_add_host_test(name [LIBS extra libraries...])
function(ppq_add_host_test name)
    cmake_parse_arguments(PARSE_ARGV 1 arg "" "" "LIBS")
    add_executable(${name} ${name}.cpp)
    target_link_libraries(${name} PRIVATE ppq_host ppq_warnings ${arg_LIBS})
    target_compile_definitions(${name} PRIVATE TEST_VECTORS_DIR="${PPQ_TEST_VECTORS_DIR}")
    add_test(NAME ${name} COMMAND ${name})
    set_tests_properties(${name} PROPERTIES LABELS host)
endfunction()

function(ppq_add_app_test name regex)
    add_test(NAME ${name} COMMAND ${ARGN})
    set_tests_properties(${name} PROPERTIES LABELS app PASS_REGULAR_EXPRESSION "${regex}")
endfunction()

# ppq_add_gpu_test(name [LIBS extra libraries...])
function(ppq_add_gpu_test name)
    cmake_parse_arguments(PARSE_ARGV 1 arg "" "" "LIBS")
    ppq_gpu_sources(${name}.hip)
    add_executable(${name} ${name}.hip)
    target_link_libraries(${name} PRIVATE ppq_core ppq_warnings ${arg_LIBS})
    target_compile_definitions(${name} PRIVATE TEST_VECTORS_DIR="${PPQ_TEST_VECTORS_DIR}")
    add_test(NAME ${name} COMMAND ${name})
    # gpu_test::run_all() exits 77 when no device is visible
    set_tests_properties(${name} PROPERTIES LABELS gpu SKIP_RETURN_CODE 77)
endfunction()
